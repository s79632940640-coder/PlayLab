Хранение данных в Supabase

Можно синхронизировать весь файл с Supabase как один JSON-документ. Это удобно для хранения резервной копии и работы с разных устройств.

Перед использованием создай таблицу club_finance_data в Supabase. SQL для создания таблицы указан ниже.
SQL для Supabase 
create table if not exists public.club_finance_data (
  club_id text primary key,
  payload jsonb not null,
  updated_at timestamptz not null default now()
);

alter table public.club_finance_data enable row level security;

create policy "Allow anon read club finance data"
on public.club_finance_data
for select
to anon
using (true);

create policy "Allow anon upsert club finance data"
on public.club_finance_data
for insert
to anon
with check (true);

create policy "Allow anon update club finance data"
on public.club_finance_data
for update
to anon
using (true)
with check (true);
